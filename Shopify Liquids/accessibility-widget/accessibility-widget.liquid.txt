{%- comment -%}
  ============================================================================
  AVRO — ACCESSIBILITY WIDGET  (Shopify Online Store 2.0 section)
  ----------------------------------------------------------------------------
  A floating, on-brand accessibility button + control panel that lets visitors
  adjust the whole storefront to their needs (text size, contrast, readable
  font, link highlighting, text spacing, reduced motion, larger cursor, and a
  reading guide). Preferences persist across pages via localStorage.

  HOW TO USE
  1. Create sections/avro-accessibility-widget.liquid and paste this file in.
  2. Render it globally so it appears on every page. In layout/theme.liquid,
     just before </body>, add:
        {% section 'avro-accessibility-widget' %}
     (or add it once to a shared section group).
  3. In the theme editor, set the "Accessibility statement" link to your
     /pages/accessibility page, pick a side (left/right), and adjust copy.

  NOTES
  - Effects are applied via classes on <html> and inline font-size overrides,
    so they cascade across the whole theme regardless of section.
  - The widget UI itself is excluded from the effects so it stays legible.
  - No external requests, no dependencies. Meets common WCAG 2.1 AA controls
    (resize text, contrast, spacing 1.4.12, reduced motion, link identification).
  ============================================================================
{%- endcomment -%}

<script>
  (function () {
    var root = document.getElementById('avro-a11y-{{ section.id }}');
    if (!root || root.dataset.a11yInit === '1') return;
    root.dataset.a11yInit = '1';

    var htmlEl = document.documentElement;
    var trigger = root.querySelector('[data-a11y-trigger]');
    var panel = root.querySelector('.a11y-panel');
    var closeBtn = root.querySelector('[data-a11y-close]');
    var guide = document.getElementById('avro-a11y-guide-{{ section.id }}');
    var STORE_KEY = 'avro-a11y-prefs';

    var TEXT_STEPS = [1, 1.15, 1.3, 1.5];
    var TEXT_SEL = 'p,li,a,span,em,strong,small,label,button,input,textarea,select,h1,h2,h3,h4,h5,h6,blockquote,figcaption,td,th,dt,dd,summary';

    var state = { textStep: 0, toggles: {} };

    /* ---------- persistence ---------- */
    function load() {
      try {
        var raw = localStorage.getItem(STORE_KEY);
        if (raw) { var p = JSON.parse(raw); state.textStep = p.textStep || 0; state.toggles = p.toggles || {}; }
      } catch (e) {}
    }
    function save() {
      try { localStorage.setItem(STORE_KEY, JSON.stringify(state)); } catch (e) {}
    }

    /* ---------- text scaling (works with px or rem themes) ---------- */
    function inWidget(el) { return root.contains(el); }

    function applyText() {
      var factor = TEXT_STEPS[state.textStep] || 1;
      if (factor === 1) { clearText(); updateTextLabel(); return; }
      var nodes = document.querySelectorAll(TEXT_SEL);
      for (var i = 0; i < nodes.length; i++) {
        var el = nodes[i];
        if (inWidget(el)) continue;
        if (!el.hasAttribute('data-a11y-fs')) {
          el.setAttribute('data-a11y-fs', parseFloat(getComputedStyle(el).fontSize) || 16);
        }
        var base = parseFloat(el.getAttribute('data-a11y-fs')) || 16;
        el.style.setProperty('font-size', (base * factor).toFixed(2) + 'px', 'important');
      }
      updateTextLabel();
    }
    function clearText() {
      var nodes = document.querySelectorAll('[data-a11y-fs]');
      for (var i = 0; i < nodes.length; i++) {
        nodes[i].style.removeProperty('font-size');
        nodes[i].removeAttribute('data-a11y-fs');
      }
    }
    function updateTextLabel() {
      var v = root.querySelector('[data-a11y-textval]');
      if (v) v.textContent = Math.round((TEXT_STEPS[state.textStep] || 1) * 100) + '%';
    }

    /* ---------- toggles ---------- */
    var CLASS_MAP = {
      contrast: 'avro-a11y-contrast', links: 'avro-a11y-links', readable: 'avro-a11y-readable',
      spacing: 'avro-a11y-spacing', motion: 'avro-a11y-motion', cursor: 'avro-a11y-cursor', guide: 'avro-a11y-guide'
    };
    function applyToggle(key) {
      var on = !!state.toggles[key];
      htmlEl.classList.toggle(CLASS_MAP[key], on);
      var btn = root.querySelector('[data-a11y-toggle="' + key + '"]');
      if (btn) btn.setAttribute('aria-pressed', on ? 'true' : 'false');
      if (key === 'motion' && on) {
        document.querySelectorAll('video[autoplay]').forEach(function (v) { try { v.pause(); } catch (e) {} });
      }
    }
    function applyAll() {
      Object.keys(CLASS_MAP).forEach(applyToggle);
      applyText();
    }

    /* ---------- open / close ---------- */
    var lastFocus = null;
    function openPanel() {
      panel.hidden = false;
      requestAnimationFrame(function () { root.classList.add('is-open'); });
      trigger.setAttribute('aria-expanded', 'true');
      lastFocus = document.activeElement;
      var first = panel.querySelector('button, a, [tabindex]');
      if (first) first.focus();
      document.addEventListener('keydown', onKey);
    }
    function closePanel() {
      root.classList.remove('is-open');
      trigger.setAttribute('aria-expanded', 'false');
      document.removeEventListener('keydown', onKey);
      setTimeout(function () { panel.hidden = true; }, 220);
      if (trigger) trigger.focus();
    }
    function onKey(e) { if (e.key === 'Escape') closePanel(); }

    /* ---------- reading guide follows pointer ---------- */
    function onMove(e) { guide.style.top = (e.clientY - 22) + 'px'; }
    function syncGuideListener() {
      if (state.toggles.guide) document.addEventListener('mousemove', onMove);
      else document.removeEventListener('mousemove', onMove);
    }

    /* ---------- events ---------- */
    trigger.addEventListener('click', function () {
      if (root.classList.contains('is-open')) closePanel(); else openPanel();
    });
    closeBtn.addEventListener('click', closePanel);
    document.addEventListener('click', function (e) {
      if (root.classList.contains('is-open') && !root.contains(e.target)) closePanel();
    });

    root.querySelectorAll('[data-a11y-toggle]').forEach(function (btn) {
      btn.addEventListener('click', function () {
        var key = btn.getAttribute('data-a11y-toggle');
        state.toggles[key] = !state.toggles[key];
        applyToggle(key);
        if (key === 'guide') syncGuideListener();
        save();
      });
    });

    root.querySelectorAll('[data-a11y-text]').forEach(function (btn) {
      btn.addEventListener('click', function () {
        var dir = btn.getAttribute('data-a11y-text');
        if (dir === 'up') state.textStep = Math.min(TEXT_STEPS.length - 1, state.textStep + 1);
        else state.textStep = Math.max(0, state.textStep - 1);
        applyText();
        save();
      });
    });

    root.querySelector('[data-a11y-reset]').addEventListener('click', function () {
      state = { textStep: 0, toggles: {} };
      Object.keys(CLASS_MAP).forEach(function (k) { htmlEl.classList.remove(CLASS_MAP[k]); });
      root.querySelectorAll('[data-a11y-toggle]').forEach(function (b) { b.setAttribute('aria-pressed', 'false'); });
      clearText();
      updateTextLabel();
      syncGuideListener();
      save();
    });

    /* ---------- init ---------- */
    load();
    root.querySelectorAll('[data-a11y-toggle]').forEach(function (btn) {
      var key = btn.getAttribute('data-a11y-toggle');
      btn.setAttribute('aria-pressed', state.toggles[key] ? 'true' : 'false');
    });
    applyAll();
    syncGuideListener();
  })();
</script>
