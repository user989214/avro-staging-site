{%- comment -%}
  ============================================================================
  AVRO — PRODUCT LIST (collection grid)
  Drop-in replacement for the theme's "Product List" section, restyled to
  match the AVRO site's product cards (Three formulas grid): rounded gray
  thumbnail, bold name, support line, divider with Best For + caffeine line,
  and an outlined pill "Shop" button.
  • Pulls real products from a Shopify collection — nothing hardcoded.
  • Per-product copy comes from metafields when set:
      - custom.support_line   → short support line under the name
      - custom.best_for       → "Best For:" description
      - custom.caffeine_line  → e.g. "No Caffeine" / "120mg Natural Caffeine"
    Cards render fine without them (rows are hidden if the metafield is empty).
  HOW TO USE
  1. Create sections/product-list-avro.liquid and paste this file in.
  2. Add the "AVRO Product List" section, pick a collection in settings.
  ============================================================================
{%- endcomment -%}

{%- assign max_items = section.settings.max_products -%}
{%- assign coll = section.settings.collection -%}

{%- case section.settings.corner_style -%}
  {%- when 'square' -%}{%- assign btn_radius = 12 -%}{%- assign card_radius = 10 -%}
  {%- when 'soft' -%}{%- assign btn_radius = 22 -%}{%- assign card_radius = 16 -%}
  {%- else -%}{%- assign btn_radius = 9999 -%}{%- assign card_radius = 20 -%}
{%- endcase -%}

<div id="avro-plist-{{ section.id }}">
  <div class="apl-wrap">

    {%- comment -%} ---------- HEADER ---------- {%- endcomment -%}
    {%- if section.settings.heading != blank or section.settings.subheading != blank -%}
      <header class="apl-head">
        <div class="apl-head-copy">
          {%- if section.settings.heading != blank -%}
            <h2 class="apl-title">{{ section.settings.heading }}</h2>
          {%- endif -%}
          {%- if section.settings.subheading != blank -%}
            <p class="apl-sub">{{ section.settings.subheading }}</p>
          {%- endif -%}
        </div>
        {%- if section.settings.show_view_all and coll != blank -%}
          <a href="{{ coll.url }}" class="apl-viewall">{{ section.settings.view_all_label | default: 'View all' }}</a>
        {%- endif -%}
      </header>
    {%- endif -%}

    {%- comment -%} ---------- GRID ---------- {%- endcomment -%}
    <div class="apl-grid">
      {%- if coll != blank -%}
        {%- for product in coll.products limit: max_items -%}
          <article class="apl-card">
            <a href="{{ product.url }}" class="apl-thumb" aria-label="{{ product.title | escape }}">
              {%- if product.featured_image -%}
                <img
                  src="{{ product.featured_image | image_url: width: 800 }}"
                  alt="{{ product.featured_image.alt | default: product.title | escape }}"
                  loading="lazy"
                  width="400"
                  height="400"
                >
              {%- endif -%}
            </a>
            <h3 class="apl-name"><a href="{{ product.url }}">{{ product.title }}</a></h3>
            {%- if product.metafields.custom.support_line != blank -%}
              <p class="apl-support">{{ product.metafields.custom.support_line }}</p>
            {%- endif -%}

            {%- assign has_bestfor = false -%}
            {%- if product.metafields.custom.best_for != blank -%}{%- assign has_bestfor = true -%}{%- endif -%}
            {%- assign has_caffeine = false -%}
            {%- if product.metafields.custom.caffeine_line != blank -%}{%- assign has_caffeine = true -%}{%- endif -%}

            {%- if has_bestfor or has_caffeine or section.settings.show_price -%}
              <div class="apl-details">
                {%- if has_bestfor -%}
                  <div>
                    <span class="apl-bestfor-lbl">{{ section.settings.best_for_label | default: 'Best For:' }}</span>
                    <p class="apl-bestfor">{{ product.metafields.custom.best_for }}</p>
                  </div>
                {%- endif -%}
                {%- if has_caffeine -%}
                  <p class="apl-caffeine">{{ product.metafields.custom.caffeine_line }}</p>
                {%- endif -%}
                {%- if section.settings.show_price -%}
                  <p class="apl-price">
                    {%- if product.compare_at_price > product.price -%}
                      <s>{{ product.compare_at_price | money }}</s>
                    {%- endif -%}
                    {{ product.price | money }}
                    {%- if product.price_varies %}+{% endif -%}
                  </p>
                {%- endif -%}
              </div>
            {%- endif -%}

            <a href="{{ product.url }}" class="apl-btn">
              {{ section.settings.shop_label | default: 'Shop' }} {{ product.title | remove: 'AVRO ' | remove: 'AVRO' | strip }}
            </a>
          </article>
        {%- endfor -%}
      {%- else -%}
        {%- comment -%} Placeholder cards until a collection is picked {%- endcomment -%}
        {%- for i in (1..max_items) limit: 3 -%}
          <article class="apl-card">
            <div class="apl-thumb" aria-hidden="true">{{ 'product-1' | placeholder_svg_tag: 'apl-placeholder' }}</div>
            <h3 class="apl-name">Product title</h3>
            <p class="apl-support">Pick a collection in the section settings.</p>
            <span class="apl-btn">{{ section.settings.shop_label | default: 'Shop' }}</span>
          </article>
        {%- endfor -%}
      {%- endif -%}
    </div>

  </div>
</div>
