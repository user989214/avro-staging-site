{%- comment -%}
  ============================================================================
  AVRO — MAIN CART (cart page)
  Drop-in replacement for the theme's main cart section, styled to match the
  AVRO site cart drawer (Your Cart + count pill, rounded thumbs, pill
  steppers, charcoal checkout button).
  • Quantity +/- and Remove use the Shopify AJAX Cart API (/cart/change.js)
    and re-render this section via the Section Rendering API — no page reload.
  • Checkout submits the native /cart form (works with selling plans,
    discounts, and Shop Pay).
  HOW TO USE
  1. Replace the contents of sections/main-cart.liquid with this file
     (or create sections/avro-cart.liquid and swap it into templates/cart.json).
  ============================================================================
{%- endcomment -%}

<script>
  (function () {
    var root = document.getElementById("avro-cart-{{ section.id }}");
    if (!root) return;
    var sectionId = root.dataset.sectionId;
    var busy = false;

    function rerender(html) {
      var parsed = new DOMParser().parseFromString(html, "text/html");
      var fresh = parsed.getElementById("avro-cart-" + sectionId);
      if (fresh) {
        root.innerHTML = fresh.innerHTML;
        bind();
      } else {
        window.location.reload();
      }
    }

    function changeLine(line, quantity) {
      if (busy) return;
      busy = true;
      root.classList.add("is-loading");
      fetch((window.Shopify && Shopify.routes && Shopify.routes.root ? Shopify.routes.root : "/") + "cart/change.js", {
        method: "POST",
        headers: { "Content-Type": "application/json", "Accept": "application/json" },
        body: JSON.stringify({ line: line, quantity: quantity, sections: sectionId })
      })
        .then(function (r) { return r.json(); })
        .then(function (data) {
          if (data && data.sections && data.sections[sectionId]) {
            rerender(data.sections[sectionId]);
          } else {
            window.location.reload();
          }
        })
        .catch(function () { window.location.reload(); })
        .finally(function () {
          busy = false;
          root.classList.remove("is-loading");
          document.dispatchEvent(new CustomEvent("cart:updated"));
        });
    }

    function bind() {
      root.querySelectorAll("[data-qty-change]").forEach(function (btn) {
        btn.addEventListener("click", function () {
          var line = parseInt(btn.dataset.line, 10);
          var delta = parseInt(btn.dataset.qtyChange, 10);
          var qEl = btn.parentElement.querySelector(".ac-q");
          var current = parseInt(qEl ? qEl.textContent : "1", 10) || 1;
          changeLine(line, Math.max(0, current + delta));
        });
      });
      root.querySelectorAll("[data-remove]").forEach(function (btn) {
        btn.addEventListener("click", function () {
          changeLine(parseInt(btn.dataset.line, 10), 0);
        });
      });
    }

    bind();
  })();
</script>
