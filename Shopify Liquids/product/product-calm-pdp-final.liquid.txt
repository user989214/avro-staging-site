{%- comment -%}
  ============================================================================
  AVRO — PRODUCT DETAIL (PDP) · CALM
  Drop-in export. Gallery images are hardcoded to the live site (one full set
  per flavor). Buy box is wired to Shopify.
  • TWO product pickers — one per flavor. Selecting a flavor switches the
    connected product: title, price, variant, subscription and gallery all swap.
  • If a flavor's product picker is left empty, it falls back to the template's
    own product.
  NOTE: Deployment Protection on the staging site must be OFF for images to load.
  HOW TO USE
  1. Create sections/product-calm.liquid and paste this file in.
  2. Add the "AVRO Product · Calm" section, then pick a product for each flavor in settings.
  ============================================================================
{%- endcomment -%}

  (function () {
    var root = document.getElementById("avro-pdp-{{ section.id }}");
    if (!root) return;

        hasProduct: {{ pa_has | json }},
        title: {{ pa.title | json }},
        variants: {{ pa.variants | json }},
        hasPlan: {{ pa_plan | json }},
        planName: {% if pa_plan %}{{ pa.selling_plan_groups.first.selling_plans.first.name | json }}{% else %}""{% endif %}
      },
      {
        key: "blackberry-jasmine",
        label: "Blackberry Jasmine",
        tokens: ["blackberry","jasmine"],
        gallery: ["https://v0-build-request-git-v0-joe-6434-a0cfbf5f-joe-6434s-projects.vercel.app/images/tubes/studio/calm-blackberry-jasmine.png","https://v0-build-request-git-v0-joe-6434-a0cfbf5f-joe-6434s-projects.vercel.app/images/tubes/solo/calm-blackberry-jasmine.jpg","https://v0-build-request-git-v0-joe-6434-a0cfbf5f-joe-6434s-projects.vercel.app/images/sticks/calm-blackberry-jasmine.png","https://v0-build-request-git-v0-joe-6434-a0cfbf5f-joe-6434s-projects.vercel.app/images/lifestyle/glass-calm-blackberry-jasmine.jpg","https://v0-build-request-git-v0-joe-6434-a0cfbf5f-joe-6434s-projects.vercel.app/images/supplement-facts/blackberry-jasmine.png"],
        hasProduct: {{ pb_has | json }},
        title: {{ pb.title | json }},
        variants: {{ pb.variants | json }},
        hasPlan: {{ pb_plan | json }},
        planName: {% if pb_plan %}{{ pb.selling_plan_groups.first.selling_plans.first.name | json }}{% else %}""{% endif %}
      }
    ];

    var vidInput = document.getElementById("pdp-vid-{{ section.id }}");
    var planInput = document.getElementById("pdp-plan-{{ section.id }}");
    var qtyEl = document.getElementById("pdp-qty-{{ section.id }}");
    var qtyInput = document.getElementById("pdp-qtyinput-{{ section.id }}");
    var addBtn = document.getElementById("pdp-add-{{ section.id }}");
    var titleEl = document.getElementById("pdp-title-{{ section.id }}");
    var galImgs = document.getElementById("gal-imgs-{{ section.id }}");
    var galThumbs = document.getElementById("gal-thumbs-{{ section.id }}");
    var addLabel = {{ section.settings.add_label | default: 'Add to Cart' | json }};

    var state = { flavor: FLAVORS[0], variant: null, buy: SHOW_SUB ? "subscribe" : "onetime", qty: 1 };

    function money(cents) { return "$" + (cents / 100).toFixed(2); }
    function planAllocation(v) {
      if (!v || !v.selling_plan_allocations || !v.selling_plan_allocations.length) return null;
      return v.selling_plan_allocations[0];
    }
    function pickVariant(fl) {
      var vs = fl.variants || [];
      if (!vs.length) return null;
      // token match on the flavor option / title first
      for (var i = 0; i < vs.length; i++) {
        var v = vs[i];
        var val = (flavorIndex > 0 ? (v["option" + flavorIndex] || "") : (v.title || "")).toString().toLowerCase();
        for (var j = 0; j < fl.tokens.length; j++) { if (val.indexOf(fl.tokens[j]) > -1) return v; }
      }
      // else first available, else first
      for (var k = 0; k < vs.length; k++) { if (vs[k].available) return vs[k]; }
      return vs[0];
    }
    function flPlanActive(fl) { return SHOW_SUB && fl.hasPlan; }

    function renderGallery(fl) {
      var urls = fl.gallery || [];
      if (!urls.length || !galImgs || !galThumbs) return;
      galImgs.innerHTML = urls.map(function (src, i) {
        return '<img data-idx="' + i + '" class="' + (i === 0 ? 'is-active' : '') + '" src="' + src + '" alt="" loading="' + (i === 0 ? 'eager' : 'lazy') + '">';
      }).join("");
      galThumbs.innerHTML = urls.map(function (src, i) {
        return '<button type="button" class="gal-thumb ' + (i === 0 ? 'is-active' : '') + '" data-target="' + i + '" aria-label="View image ' + (i + 1) + '"><img src="' + src + '" alt=""></button>';
      }).join("");
      bindThumbs();
    }
    function bindThumbs() {
      root.querySelectorAll(".gal-thumb").forEach(function (btn) {
        btn.addEventListener("click", function () {
          var idx = btn.dataset.target;
          galImgs.querySelectorAll("img").forEach(function (img) { img.classList.toggle("is-active", img.dataset.idx == idx); });
          root.querySelectorAll(".gal-thumb").forEach(function (t) { t.classList.toggle("is-active", t === btn); });
        });
      });
    }

    function render() {
      var fl = state.flavor, v = state.variant;
      root.querySelectorAll("[data-price-onetime]").forEach(function (el) { el.textContent = v ? money(v.price * state.qty) : ""; });
      var alloc = planAllocation(v);
      root.querySelectorAll("[data-price-subscribe]").forEach(function (el) { el.textContent = alloc ? money(alloc.price * state.qty) : (v ? money(v.price * state.qty) : ""); });

      // Subscribe availability for this flavor's product
      var subOpt = root.querySelector("[data-subscribe-opt]");
      var oneOpt = root.querySelector('[data-buy="onetime"]');
      var planNameEl = root.querySelector("[data-plan-name]");
      if (subOpt) {
        if (flPlanActive(fl)) {
          subOpt.style.display = "";
          if (planNameEl) planNameEl.textContent = fl.planName || "";
        } else {
          subOpt.style.display = "none";
          if (state.buy === "subscribe") {
            state.buy = "onetime";
            subOpt.classList.remove("is-active");
            if (oneOpt) oneOpt.classList.add("is-active");
          }
        }
      }

      vidInput.value = v ? v.id : "";
      if (state.buy === "subscribe" && alloc) { planInput.disabled = false; planInput.value = alloc.selling_plan.id; }
      else { planInput.value = ""; planInput.disabled = true; }

      var available = v && v.available;
      addBtn.disabled = !v || !available;
      addBtn.textContent = (!v) ? "Unavailable" : (available ? addLabel : "Sold Out");

      var benefits = root.querySelector("[data-subscribe-benefits]");
      if (benefits) benefits.style.display = (state.buy === "subscribe") ? "flex" : "none";
    }

    function applyFlavor(fl) {
      state.flavor = fl;
      state.variant = pickVariant(fl);
      if (titleEl && fl.title) titleEl.textContent = fl.title;
      renderGallery(fl);
      render();
    }

    root.querySelectorAll(".flavor").forEach(function (btn) {
      btn.addEventListener("click", function () {
        var key = btn.dataset.flavorKey;
        var fl = FLAVORS.filter(function (f) { return f.key === key; })[0];
        if (!fl) return;
        root.querySelectorAll(".flavor").forEach(function (b) { b.classList.remove("is-active"); b.setAttribute("aria-checked", "false"); });
        btn.classList.add("is-active"); btn.setAttribute("aria-checked", "true");
        applyFlavor(fl);
      });
    });

    root.querySelectorAll(".buy-opt").forEach(function (btn) {
      btn.addEventListener("click", function () {
        if (btn.style.display === "none") return;
        root.querySelectorAll(".buy-opt").forEach(function (b) { b.classList.remove("is-active"); });
        btn.classList.add("is-active");
        state.buy = btn.dataset.buy;
        render();
      });
    });

    root.querySelectorAll(".stepper button").forEach(function (btn) {
      btn.addEventListener("click", function () {
        var step = parseInt(btn.dataset.step, 10);
        state.qty = Math.max(1, state.qty + step);
        qtyEl.textContent = state.qty; qtyInput.value = state.qty;
        root.querySelector('.stepper button[data-step="-1"]').disabled = state.qty <= 1;
        render();
      });
    });

    applyFlavor(FLAVORS[0]);
  })();
</script>
